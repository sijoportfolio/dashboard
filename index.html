/** * WebApp.gs * ------------------------------------------------------------ * The bridge between everything built so far and an actual * dashboard. Deploying this as a Web App (Deploy -> Manage * deployments -> New version) gives you a URL that returns real * holdings data as JSON, and now also accepts the two actions the * ticker-card edit button needs. * * doGet handles both reads (?action=holdings etc) and the write * actions the dashboard needs - kept as GET/query-param calls (not * a separate doPost) to match the existing thin, single-file style, * since this is a personal single-user tool with no auth layer. * Kept deliberately thin - it just calls the same tested * functions everything else uses, never duplicates logic. * * Now also routes the Add Transaction form's actions (get_accounts_ * for_form, get_all_assets_for_form, get_held_assets_for_account, * get_current_quantity, get_fx_rate_for_form, create_transaction) - * these call straight through to CreateTransaction.gs, same * "thin router, no duplicated logic" rule as everything else here. * Same pattern for the watchlist actions (get_watchlist, * add_watchlist_item, remove_watchlist_item), routed straight * through to Watchlist.gs. * * CACHING: the 'holdings' action (which the dashboard's initial * load and every 5-min auto-poll call) is now backed by a short * CacheService TTL, so repeat loads within that window return * instantly instead of re-running the full FIFO lot-matching * engine over 800+ transactions every single time. Every action * that actually CHANGES data (refresh, set_manual_price, * resume_auto_fetch, create_transaction) invalidates the cache * first, so nothing ever serves stale data after a real change - * caching only saves work on repeat reads of UNCHANGED data. * ------------------------------------------------------------ */ const HOLDINGS_CACHE_KEY = 'holdings_payload_v1'; const HOLDINGS_CACHE_TTL_SECONDS = 180; // 3 min - shorter than the 5-min auto-poll, // so most poll cycles still get a fresh // compute, but rapid repeat loads (opening // the dashboard in two tabs, a quick // reload) hit cache instead of recomputing. function getCachedHoldingsPayload_() { try { const cached = CacheService.getScriptCache().get(HOLDINGS_CACHE_KEY); return cached ? JSON.parse(cached) : null; } catch (e) { return null; // cache read failing should never break the dashboard - fall through to a fresh compute } } function setCachedHoldingsPayload_(payload) { try { CacheService.getScriptCache().put(HOLDINGS_CACHE_KEY, JSON.stringify(payload), HOLDINGS_CACHE_TTL_SECONDS); } catch (e) { // CacheService rejects values over 100KB - if the holdings payload ever grows // past that (a lot more lots/holdings down the line), just skip caching that // round rather than erroring the whole request. Always-fresh is a safe fallback. Logger.log('Holdings payload too large to cache (or cache write failed): ' + e.message); } } function invalidateHoldingsCache_() { try { CacheService.getScriptCache().remove(HOLDINGS_CACHE_KEY); } catch (e) { /* nothing to do if this fails - TTL will still expire it */ } } const TXNS_CACHE_KEY = 'transactions_payload_v1'; const TXNS_CACHE_TTL_SECONDS = 180; // same window as holdings - both change together (a new transaction affects both) function getCachedTransactionsPayload_() { try { const cached = CacheService.getScriptCache().get(TXNS_CACHE_KEY); return cached ? JSON.parse(cached) : null; } catch (e) { return null; } } function setCachedTransactionsPayload_(payload) { try { CacheService.getScriptCache().put(TXNS_CACHE_KEY, JSON.stringify(payload), TXNS_CACHE_TTL_SECONDS); } catch (e) { Logger.log('Transactions payload too large to cache (or cache write failed): ' + e.message); } } function invalidateTransactionsCache_() { try { CacheService.getScriptCache().remove(TXNS_CACHE_KEY); } catch (e) { /* TTL will still expire it */ } } const RETURNS_CACHE_KEY = 'returns_payload_v1'; const RETURNS_CACHE_TTL_SECONDS = 900; // 15 min - a return % doesn't need to track live price ticks function getCachedReturnsPayload_() { try { const cached = CacheService.getScriptCache().get(RETURNS_CACHE_KEY); return cached ? JSON.parse(cached) : null; } catch (e) { return null; } } function setCachedReturnsPayload_(payload) { try { CacheService.getScriptCache().put(RETURNS_CACHE_KEY, JSON.stringify(payload), RETURNS_CACHE_TTL_SECONDS); } catch (e) { Logger.log('Returns payload too large to cache (or cache write failed): ' + e.message); } } function invalidateReturnsCache_() { try { CacheService.getScriptCache().remove(RETURNS_CACHE_KEY); } catch (e) { /* TTL will still expire it */ } } // Actions that CHANGE data - blocked entirely for the family (read-only) key, // enforced here on the server, never just hidden in the dashboard UI. Anyone // calling the URL directly with a family key still can't trigger any of these. const WRITE_ACTIONS = ['refresh', 'set_manual_price', 'resume_auto_fetch', 'create_transaction', 'add_watchlist_item', 'remove_watchlist_item']; /** * Checks e.parameter.key against config.access_key_owner / access_key_family * (stored in the Sheet, same "single source of truth" pattern as everything * else - rotate a key by editing one cell, no code redeploy needed). Returns * { ok, level, error } - level is 'owner' or 'family', used below to gate * WRITE_ACTIONS. If neither config key is set yet, the gate is OPEN (returns * ok:true, level:'owner') so this can be deployed without instantly locking * out the dashboard - fill in both config rows to actually turn the gate on. */ function checkAccessKey_(e) { const ownerKey = getConfigValue('access_key_owner', null); const familyKey = getConfigValue('access_key_family', null); if (!ownerKey && !familyKey) { return { ok: true, level: 'owner' }; // gate not configured yet - see note above } const provided = e && e.parameter && e.parameter.key; if (!provided) return { ok: false, error: 'Access key required.' }; if (ownerKey && provided === ownerKey) return { ok: true, level: 'owner' }; if (familyKey && provided === familyKey) return { ok: true, level: 'family' }; return { ok: false, error: 'Invalid access key.' }; } function doGet(e) { const action = (e && e.parameter && e.parameter.action) || 'holdings'; const access = checkAccessKey_(e); if (!access.ok) { return jsonResponse({ error: access.error }, 401); } if (access.level === 'family' && WRITE_ACTIONS.indexOf(action) !== -1) { return jsonResponse({ error: 'This key is read-only - ' + action + ' is not permitted.' }, 403); } try { let payload; if (action === 'holdings') { const cached = getCachedHoldingsPayload_(); if (cached) { payload = cached; } else { const result = CALCULATE_HOLDINGS(); payload = { asOfDate: result.asOfDate, holdingsSummary: result.holdingsSummary, holdingsLots: result.holdingsLots, warnings: result.warnings, }; setCachedHoldingsPayload_(payload); } } else if (action === 'refresh') { invalidateHoldingsCache_(); // prices are about to change - never serve the old cached payload after this REFRESH_PRICES(); const result = CALCULATE_HOLDINGS(); payload = { asOfDate: result.asOfDate, holdingsSummary: result.holdingsSummary, holdingsLots: result.holdingsLots, warnings: result.warnings, refreshedPrices: true, }; setCachedHoldingsPayload_(payload); // this fresh result becomes the new cache too, not just this response } else if (action === 'set_manual_price') { const assetId = e.parameter.asset_id; const price = Number(e.parameter.price); if (!assetId || isNaN(price)) { return jsonResponse({ error: 'set_manual_price requires asset_id and a numeric price' }, 400); } invalidateHoldingsCache_(); SET_MANUAL_PRICE(assetId, price); const result = CALCULATE_HOLDINGS(); payload = { asOfDate: result.asOfDate, holdingsSummary: result.holdingsSummary, holdingsLots: result.holdingsLots, warnings: result.warnings, }; setCachedHoldingsPayload_(payload); } else if (action === 'resume_auto_fetch') { const assetId = e.parameter.asset_id; if (!assetId) return jsonResponse({ error: 'resume_auto_fetch requires asset_id' }, 400); invalidateHoldingsCache_(); RESUME_AUTO_FETCH(assetId); const result = CALCULATE_HOLDINGS(); payload = { asOfDate: result.asOfDate, holdingsSummary: result.holdingsSummary, holdingsLots: result.holdingsLots, warnings: result.warnings, }; setCachedHoldingsPayload_(payload); } else if (action === 'targets') { payload = { targets: sheetToObjects('targets') }; } else if (action === 'transactions') { // Now cached (same pattern as holdings) - this action fires on every page // load AND every 5-min auto-poll, but the transaction ledger itself only // changes when a real BUY/SELL/etc happens, so most of those reads were // pure waste. Also drops rows with no action set (blank placeholder rows // left over from an old import-undo, ~121 of them) before sending - // they're invisible no-ops to CalcEngine's own run() but there's no // reason to ship dead weight over the wire to the dashboard too. const cachedTxns = getCachedTransactionsPayload_(); if (cachedTxns) { payload = cachedTxns; } else { const txns = sheetToObjects('transactions') .filter(t => t.action) // drops the ~121 blank/no-action rows .map(t => ({ date: t.date, action: t.action, base_currency_amount: t.base_currency_amount, asset_id: t.asset_id, account_id: t.account_id, quantity: t.quantity, price: t.price, trade_currency: t.trade_currency, fees: t.fees, })); payload = { transactions: txns }; setCachedTransactionsPayload_(payload); } } else if (action === 'returns') { // XIRR + TWR etc from CalculateReturns.gs. Cached separately from // holdings, on a longer TTL - a return % doesn't need to update on every // 5-min poll or every price refresh the way live prices do, and // CALCULATE_RETURNS is real work (dozens of in-memory valuations across // the tracking period), so recomputing it that often would be pure waste. // Only invalidated by create_transaction (a real cash-flow/holdings // change) - NOT by refresh/set_manual_price, since a live price tick // moving the current value by a few dollars doesn't meaningfully change // an ITD return percentage. const cachedReturns = getCachedReturnsPayload_(); if (cachedReturns) { payload = cachedReturns; } else { payload = CALCULATE_RETURNS(e.parameter.includeClosedGains === 'true'); setCachedReturnsPayload_(payload); } } else if (action === 'position_transactions') { // Kept for backward compatibility / other callers, but the dashboard no // longer calls this - see the transactions action above. const assetId = e.parameter.asset_id; const accountId = e.parameter.account_id; if (!assetId || !accountId) { return jsonResponse({ error: 'position_transactions requires asset_id and account_id' }, 400); } const txns = sheetToObjects('transactions') .filter(t => t.asset_id === assetId && t.account_id === accountId) .map(t => ({ date: t.date, action: t.action, base_currency_amount: t.base_currency_amount, quantity: t.quantity, price: t.price, trade_currency: t.trade_currency, fees: t.fees, })); payload = { transactions: txns }; } else if (action === 'accounts') { payload = { accounts: sheetToObjects('accounts') }; } else if (action === 'config') { payload = { config: getConfigMap() }; } else if (action === 'get_accounts_for_form') { payload = { accounts: GET_ACCOUNTS_FOR_FORM() }; } else if (action === 'get_all_assets_for_form') { payload = { assets: GET_ALL_ASSETS_FOR_FORM() }; } else if (action === 'get_held_assets_for_account') { const accountId = e.parameter.account_id; if (!accountId) return jsonResponse({ error: 'get_held_assets_for_account requires account_id' }, 400); payload = { assets: GET_HELD_ASSETS_FOR_ACCOUNT(accountId) }; } else if (action === 'get_current_quantity') { const accountId = e.parameter.account_id; const assetId = e.parameter.asset_id; if (!accountId || !assetId) return jsonResponse({ error: 'get_current_quantity requires account_id and asset_id' }, 400); payload = { quantity: GET_CURRENT_QUANTITY(accountId, assetId) }; } else if (action === 'get_fx_rate_for_form') { const fromCcy = e.parameter.from_currency; const toCcy = e.parameter.to_currency; if (!fromCcy || !toCcy) return jsonResponse({ error: 'get_fx_rate_for_form requires from_currency and to_currency' }, 400); payload = GET_FX_RATE_FOR_FORM(fromCcy, toCcy, e.parameter.date); } else if (action === 'get_watchlist') { payload = { watchlist: GET_WATCHLIST() }; } else if (action === 'add_watchlist_item') { payload = ADD_WATCHLIST_ITEM(e.parameter); } else if (action === 'remove_watchlist_item') { const watchlistId = e.parameter.watchlist_id; if (!watchlistId) return jsonResponse({ error: 'remove_watchlist_item requires watchlist_id' }, 400); payload = REMOVE_WATCHLIST_ITEM(watchlistId); } else if (action === 'create_transaction') { // Changes holdings (a new row affects quantities/cash/gains) - invalidate // BEFORE writing, same rule every other write action in this file follows, // so the very next 'holdings' load can never serve a stale cached payload. // Also the only action that invalidates transactions/returns caches - a // new ledger row is the only thing that actually changes either of those. invalidateHoldingsCache_(); invalidateTransactionsCache_(); invalidateReturnsCache_(); payload = CREATE_TRANSACTION(e.parameter); } else { return jsonResponse({ error: 'Unknown action: ' + action }, 400); } if (payload && typeof payload === 'object') payload.accessLevel = access.level; return jsonResponse(payload, 200); } catch (err) { return jsonResponse({ error: err.message }, 500); } } function jsonResponse(obj, statusCode) { const output = ContentService.createTextOutput(JSON.stringify(obj)); output.setMimeType(ContentService.MimeType.JSON); return output; }
